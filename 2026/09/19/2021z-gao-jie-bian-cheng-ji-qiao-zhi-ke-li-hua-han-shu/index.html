什么是高阶函数

  1. 一个函数的参数是一个函数，我们就可以称该函数为高阶函数「回调函数」
  2. 一个函数返回一个函数，我们也称之为高阶函数「不单指闭包」
function fn() {
    var a = 1;

    return function() {
        console.log(a);
    }
}

// 这样其实执行完 两个函数都被释放了，严格意义上来讲，没有形成闭包「没有持续引用」。
fn()();

体会下高阶函数实现的 AOP

function coreFn(a, b, c) {
    console.log('core code is running', a, b, c);
}

// 我们想在 coreFn 执行之前做一些事情
Function.prototype.before = function (beforeFn) {
    return (...args) => {
        beforeFn(); // do something
        this(...args); // 执行原函数
    }
}

var newFn = coreFn.before(() => {
    console.log('执行插入的代码');
});

newFn(1, 2, 3); // "执行插入的代码"      "core code is running 1 2 3"

抛砖引玉

// 怎么实现 fn 能输出 6, 参数个数可能是多个哦

let res = fn(1, 2)(3);

console.log(res); // 1 + 2 + 3 => 6

实现方式

const fn = (...params) => {
    return (...args) => {
        params = params.concat(args); // 合并参数

        return params.reduce((result, item) => result + item); // 求和
    }
}

以上这种做法，其实就是柯里化思想的实践，只不过较为简单。

什么是柯里化函数

柯里化，英语：Currying(果然是满满的英译中的既视感)。

柯里化函数：预先存储(预处理)，利用闭包的”保存机制”，我们把一些值事先储存起来，供其下级上下文调用，这种把接受多个参数的函数根据参数的个数转化成 n 个函数，叫做柯里化，柯里化我们一般都认为参数是一个一个传递的。

如果多个参数的函数处理后还是可以接受非单一参数呢？这是偏函数。

// 柯里化函数的通用实现
function curring(fn){
    const inner = (args = []) =>{ 
        return args.length >= fn.length ? 
               fn(...args) : 
               (...arr)=> inner([...args,...arr]);
    }

    return inner();
}

柯里化实践

我们来实现一个较为复杂的柯里化方法，又能调用多次(每次 return 一个新函数)，而打印 return 结果时，又打印出数次传承累加的结果。

const curring = function curring() {
    let params = []; // 缓存每次传入的值到 curring AO 中

    const add = (...args) => {
        params = params.concat(args);

        return add;
    }

    // 利用了 console.log 打印函数 会把函数隐式转 string 的机制。
    // 执行顺序为 [Symbol.toPrimitive] --> valueOf --> toString
    add[Symbol.toPrimitive] = () => params.reduce((prev, next) => prev + next)
    return add;
};

let add = curring();
let res = add(1)(2)(3);
console.log(res); // => 6

add = curring();
res = add(1, 2, 3)(4);
console.log(res); // => 10

add = curring();
res = add(1)(2)(3)(4)(5);
console.log(res); // => 15

柯里化思想解决函数组合洋葱式调用问题

在函数时编程当中，有一个很重要的概念就是函数组合，实际上就是把处理数据的函数，像管道一样连接起来，然后让数据穿过管道得到最终的结果，例如

const add1 = x => x + 1;
const mul3 = x => x * 3;
const div2 = x => x / 2;

div2(mul3(add1(add1(0)))) // => 3

而这样的写法可读性明显太差了，我们可以构建一个 componse 函数，它接受任意多个函数作为参数(这些函数都只接收一个参数)，然后 compose 返回的也是一个函数，达到以下的效果：

const operate = compose(div2, mul3, add1, add1);

operate(0) // => 相当于 div2(mul3(add1(add1(0))))
operate(2) // => 相当于 div2(mul3(add1(add1(2))))

简而言之：compose 方法可以把类似 f(g(h(x))) 这种写法简化成 compose(f, g, h)(x), 请你完成 compose 函数的编写。

const add1 = x => x + 1;
const mul3 = x => x * 3;
const div2 = x => x / 2;

const compose = function compose(...funcs) {
    let len = funcs.length;

    // reduce 如果目标数组为空 会报错
    if (len === 0) return x => x;

    // 这里直接返回函数即可 返回函数则 opreat = returnFn，opreat(0) 调用的时候也会执行。
    if (len === 1) return x => funcs[0]); 

    // funcs: [div2, mul3, add1, add1]
    return function operate(x) {
        //  x: 0
        // 这里要实现把 funcs 从右到左挨个执行一遍，并把前一个函数执行结果传递给后一个函数
        // 所以 reduce 搞起来, 每次的计算结果 next(x) 当成下一次的 prev 继续传递执行
        return funcs.reduceRight((prev, next) => next(prev), x);
    }
}

const operate = compose(div2, mul3, add1, add1);
console.log(operate(0));

redux 中的柯里化思想

redux 中也有一个 compose 函数，也是柯里化的思想，这里跟上面差不多，可以了解下~

function compose(...funcs) {
    if (funcs.length === 0) {
        return arg => arg
    }

    if (funcs.length === 1) {
        return funcs[0]
    }

    return funcs.reduce((a, b) => (...args) => a(b(...args)));
}
