默认启用的是 hash 模式的路由。

Vue-router 支持的路由模式

针对前端而言，路由就是访问不同的路径，返回不同的组件，Vue 中的路由分为两种：

  • hash 模式，路径后面带 /#/a
  • historyApi 默认不带 /a
  • memoryHistory 不会发生路径变化，但是可以切换组件(微前端中，我们希望不显示子应用路径，可以使用这种路由，也会保存历史哦)

由于早期实现跳转都是后端来做的，spa 应用不用向服务器发请求可以直接进行跳转，乍一看，hash 路由满足它的条件。

hash 路由切换是不向服务器发送请求的，但是呢，它太丑了，而且服务器无法获取 hash 值(不发请求)，从而没办法做 SEO 优化，它的优点是方便，不需要服务端支持。而 history 路由优点就是漂亮啦，而且支持服务端渲染，可以做 SEO 优化，需要服务端支持(如果访问的资源不存在，我们可以跳转到首页，渲染首页内容，渲染的时候前端会取到路径渲染组件)。

code 
import RouterLink from "./components/RouterLink";
import RouterView from "./components/RouterView";
import { createMatcher } from "./create-matcher";
import HashHistory from "./history/hash";
import BrowserHistory from "./history/history";
export let Vue;

class VueRouter { // 稍后new的时候会产生一个路由实例
  constructor(options = {}) {
    // 1. matcher 中有 动态添加路由 和 路由的匹配方法 { addRoutes, match }
    this.matcher = createMatcher(options.routes || []); // 调用匹配器

    // 2. 默认有两种路由，hash history(不去考虑 memoryHistory)
    if (options.mode === 'hash') {
      this.history = new HashHistory(this);
    } else {
      this.history = new BrowserHistory(this);
    }

    // this.matcher.match 匹配规则， this.history 路由系统
  }


  match(location) { // 访问 match 的时候，少写一层，代理
    return this.matcher.match(location);
  }

  init(app) { // 路由初始化 监听路径变换，渲染对应组件
    const history = this.history;

    history.listen((route) => { // _route就是我们定义的响应式数据
      app._route = route; // 安插的回调每次都重新 渲染一下
    })

    // 老版本中 hash 监控靠的是hashchange 新版本可以使用popstate
    const setupListenerHandler = () => {
      history.setupListener(); // 监听hash值的变化 
    }

    // 根据路径切换页面内容 ，并且跳转完毕后 监听hash值的变化，核心方法
    history.transitionTo(history.getCurrentLocation(), setupListenerHandler)
  }

  push(location) {
    // 跳转到这个路径
    this.history.transitionTo(location, () => { // 根据路径切换页面内容
        this.history.updateLocation(location) // 更新页面路径
    });
  }
}

VueRouter.install = (_Vue) => {
  Vue = _Vue; // 全局保存 Vue 类，在其他文件中使用
  // 全局注册 router-link 组件
  Vue.component('router-link', RouterLink);
  // 全局注册 router-view 组件
  Vue.component('router-view', RouterView);

  Vue.mixin({
    beforeCreate() {
      // 需要将根的 router 对象共享给每个组件
      if (this.$options.router) { // 根组件
        // console.log('root', this.$options.router);
        this._routerRoot = this; // 将自己暴露出来
        this._router = this.$options.router; // 在自己的身上添加一个_router属性

        this._router.init(this); // 路由初始化
        // Vue.util.defineReactive(目标对象, 目标对象属性, 属性值),
        // 将 this._route 定义为一个响应式数据
        // 每次 current 变，要更改 this._route 的值(history.listen)
        Vue.util.defineReactive(this, '_route', this._router.history.current);
        console.warn(this);
      } else { // 子组件
        // console.log('child', this.$options.name);
        // 以后子组件可以通过 this._routerRoot._router 拿到 roouter 实例
        // 共享的是上级组件的实例
        this._routerRoot = this.$parent && this.$parent._routerRoot;
      }
    }
  });

  // 方便组件访问 current, {path:'/',matched:[],params,query}
  Object.defineProperty(Vue.prototype, '$route', {
    get() { // 注意是 _route，也就是 this._router.history.current
      return this._routerRoot._route;
    }
  })

  // 方便组件访问 router 的实例
  Object.defineProperty(Vue.prototype, '$router', { // 代理就是取值的时候才调用get方法  vm._data.xxx => vm.xxx
    get() {
      return this._routerRoot._router; // match  push replace
    }
  })
}

export default VueRouter

路由解析器 src/ys-vue-router/create-matcher.js

code 
import { createRouteMap } from "./creater-route-map";
export const createRoute = (record, { path }) => {
    let matched = [];
    if (record){
        while (record) { // 有记录将自己插入到队列的头部
            matched.unshift(record);
            record = record.parent; // 一层层的向上找
        }
    }
    return {
        path,
        matched
    }
}
export function createMatcher(routes) { // 匹配器 
    const { pathMap } = createRouteMap(routes); // 创建一个 路径和记录的映射表
    function addRoutes(routes) { // 在原来的基础上继续添加路由
        createRouteMap(routes, pathMap);
    }

    // { path: 'about/a', matched: [about组件, a组件] }
    function match(location) { // 给我传递个路径 我帮你去匹配所有的组件
        let record = pathMap[location]; 
        if (record) { // 注册过
            return createRoute(record, { path: location })
        }
        return createRoute(null, { path: location })
    }

    // console.log(match('/about/a'));
    // 一个路径对应的记录是谁，这个记录有没有父亲 有父亲也要创造一个关联
    return {
        addRoutes, 
        match
    }
}

hash&historyApi 的基类，src/ys-vue-router/history/base.js

code 
import { createRoute } from "../create-matcher";

class Base {
  constructor(router) {
    this.router = router;
    // 我的想法是将current属性变成响应式的，如果在渲染router-view时候用到了这个current
    // 等会current变化了就可以重新刷新视图
    this.current = createRoute(null, { path: '/' })
  }

  listen(cb){ // 更改 this._route 数据，触发页面更新
    this.cb = cb;
  }

  // 根据路径 拿到对应的record，并监听 hash 变化
  // 每次hash改变，重新走这里，但不传监听函数啦(改传更新url的方法了)
  transitionTo(location, callback) { // 根据路径进行跳转
    // 根据路径进行匹配 ，匹配到对应的记录
    // { path: '/about/a', matched: [about组件, a组件] }
    let record = this.router.match(location); 

    // 重新计算 current
    this.current = createRoute(record, { path: location });

    console.error(this.current);
    // 下面这句 this._route = current; 重新赋值，触发更新
    this.cb && this.cb(this.current);
    // 每次record 发生变化 需要渲染页面 （响应式数据）

    callback && callback(); // callback默认只有第一次传递
  }
}

export default Base


hash 路由类，src/ys-vue-router/history/hash.js

code 
import Base from "./base";

function ensureHash() {
  if (window.location.hash) return

  window.location.hash = '/' // 给页面增加 /#/
}


class HashHistroy extends Base {
  constructor(router) {
    super(router);
    // 确保刷新的时候有hash #
    ensureHash()
  }
  getCurrentLocation() { // 获取当前的路径
    return window.location.hash.slice(1)
  }
  setupListener() { // 监听 hash 变化
    window.addEventListener('hashchange', () => {
      this.transitionTo(this.getCurrentLocation())
    })
  }
  updateLocation(location) {
    window.location.hash = location
  }
}

export default HashHistroy


history路由类，vue-router-demo/src/ys-vue-router/history/history.js

import Base from "./base";

class BrowserHistory extends Base {
    constructor(router) {
        super(router);
    }
    getCurrentLocation(){
        return window.location.pathname 
    }
    updateLocation(location) {
        history.pushState({},null,location);
    }
    setupListener() {
        // 监听浏览器的前进后退事件
        window.addEventListener('popstate', (e) => {
            this.transitionTo(this.getCurrentLocation()); // 重新跳转
        })
    }
}
export default BrowserHistory
